## New File: src/scanner/mod.rs

```rust
mod discovery;
mod validation;
mod output;

pub use discovery::*;
pub use output::*;
pub use validation::*;

use std::path::PathBuf;

/// Result of a scan operation
#[derive(Debug)]
pub struct ScanResult {
    pub skills: Vec<ScannedSkill>,
    pub duplicates: Vec<DuplicateSkill>,
    pub scan_errors: Vec<ScanError>,
}

impl ScanResult {
    pub fn new() -> Self {
        Self {
            skills: Vec::new(),
            duplicates: Vec::new(),
            scan_errors: Vec::new(),
        }
    }
    
    pub fn is_empty(&self) -> bool {
        self.skills.is_empty() && self.scan_errors.is_empty()
    }
    
    pub fn valid_count(&self) -> usize {
        self.skills.iter().filter(|s| s.validation.errors.is_empty()).count()
    }
    
    pub fn invalid_count(&self) -> usize {
        self.skills.iter().filter(|s| !s.validation.errors.is_empty()).count()
    }
    
    pub fn warning_count(&self) -> usize {
        self.skills.iter().map(|s| s.validation.warnings.len()).sum()
    }
}

/// A scanned and optionally validated skill
#[derive(Debug, Clone)]
pub struct ScannedSkill {
    pub name: String,
    pub path: PathBuf,
    pub validation: crate::validator::ValidationResult,
    pub tool: Option<String>, // Which tool config found this skill
}

/// A duplicate skill found in multiple locations
#[derive(Debug, Clone)]
pub struct DuplicateSkill {
    pub name: String,
    pub paths: Vec<PathBuf>,
}

/// Error during scanning (not validation error)
#[derive(Debug, Clone)]
pub struct ScanError {
    pub path: PathBuf,
    pub error: String,
}

/// Output format for scan results
#[derive(Debug, Clone, Copy)]
pub enum OutputFormat {
    Summary,
    Verbose,
    Json,
}
```

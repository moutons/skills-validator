mod helpers;
use helpers::scan_test_context::ScanTestContext;

#[test]
fn test_scan_user_mode_finds_home_skills() {
    let ctx = ScanTestContext::new();
    
    // Create skills in home directory
    ctx.create_skill(ctx.home(), ".claude/skills", "home-skill-1");
    ctx.create_skill(ctx.home(), ".claude/skills", "home-skill-2");
    
    // Run scan --user
    let output = ctx.run_scan(&["--user"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("home-skill-1"));
    assert!(output.stdout.contains("home-skill-2"));
    assert!(output.stdout.contains("Found 2 skills"));
}

#[test]
fn test_scan_repo_mode_requires_git_repo() {
    let ctx = ScanTestContext::new();
    
    // Don't init git repo
    
    // Run scan --repo
    let output = ctx.run_scan(&["--repo"]);
    
    assert!(!output.success());
    assert_eq!(output.code(), Some(2));
    assert!(output.stderr.contains("Not in a git repository"));
}

#[test]
fn test_scan_repo_mode_finds_repo_skills() {
    let mut ctx = ScanTestContext::new();
    ctx.init_git_repo();
    
    // Create skills in repo
    ctx.create_skill(ctx.repo(), ".claude/skills", "repo-skill");
    
    // Run scan --repo
    let output = ctx.run_scan(&["--repo"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("repo-skill"));
}

#[test]
fn test_scan_all_mode_combines_repo_and_home() {
    let mut ctx = ScanTestContext::new();
    ctx.init_git_repo();
    
    // Create skills in both locations
    ctx.create_skill(ctx.home(), ".claude/skills", "home-skill");
    ctx.create_skill(ctx.repo(), ".claude/skills", "repo-skill");
    
    // Run scan --all
    let output = ctx.run_scan(&["--all"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("home-skill"));
    assert!(output.stdout.contains("repo-skill"));
    assert!(output.stdout.contains("Found 2 skills"));
}

#[test]
fn test_scan_tool_mode_filters_by_tool() {
    let mut ctx = ScanTestContext::new();
    ctx.init_git_repo();
    
    // Create skills for different tools
    ctx.create_skill(ctx.home(), ".claude/skills", "claude-skill");
    ctx.create_skill(ctx.home(), ".cursor/skills", "cursor-skill");
    
    // Run scan --tool claude-code
    let output = ctx.run_scan(&["--tool", "claude-code"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("claude-skill"));
    assert!(!output.stdout.contains("cursor-skill"));
}

#[test]
fn test_scan_tool_mode_multiple_tools() {
    let ctx = ScanTestContext::new();
    
    ctx.create_skill(ctx.home(), ".claude/skills", "claude-skill");
    ctx.create_skill(ctx.home(), ".cursor/skills", "cursor-skill");
    ctx.create_skill(ctx.home(), ".opencode/skills", "opencode-skill");
    
    // Run scan --tool claude-code,cursor
    let output = ctx.run_scan(&["--tool", "claude-code,cursor"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("claude-skill"));
    assert!(output.stdout.contains("cursor-skill"));
    assert!(!output.stdout.contains("opencode-skill"));
}

#[test]
fn test_scan_invalid_tool_shows_available_tools() {
    let ctx = ScanTestContext::new();
    
    // Run scan --tool nonexistent
    let output = ctx.run_scan(&["--tool", "nonexistent"]);
    
    assert!(!output.success());
    assert!(output.stderr.contains("claude-code"));
    assert!(output.stderr.contains("cursor"));
    assert!(output.stderr.contains("opencode"));
}

#[test]
fn test_scan_dry_run_skips_validation() {
    let ctx = ScanTestContext::new();
    
    // Create invalid skill
    ctx.create_invalid_skill(ctx.home(), ".claude/skills", "invalid-skill");
    
    // Run with --dry-run
    let output = ctx.run_scan(&["--user", "--dry-run"]);
    
    assert!(output.success()); // Should succeed even with invalid skills
    assert!(output.stdout.contains("invalid-skill"));
}

#[test]
fn test_scan_validates_skills() {
    let ctx = ScanTestContext::new();
    
    // Create valid and invalid skills
    ctx.create_skill(ctx.home(), ".claude/skills", "valid-skill");
    ctx.create_invalid_skill(ctx.home(), ".claude/skills", "invalid-skill");
    
    let output = ctx.run_scan(&["--user"]);
    
    assert!(!output.success());
    assert_eq!(output.code(), Some(1));
    assert!(output.stdout.contains("Valid: 1"));
    assert!(output.stdout.contains("Invalid: 1"));
}

#[test]
fn test_scan_detects_duplicates() {
    let mut ctx = ScanTestContext::new();
    ctx.init_git_repo();
    
    // Create same skill in both locations
    ctx.create_skill(ctx.home(), ".claude/skills", "duplicate-skill");
    ctx.create_skill(ctx.repo(), ".claude/skills", "duplicate-skill");
    
    let output = ctx.run_scan(&["--all"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("duplicate"));
    assert!(output.stdout.contains("duplicate-skill"));
}

#[test]
fn test_scan_no_skills_found_warning() {
    let ctx = ScanTestContext::new();
    
    // Don't create any skills
    
    let output = ctx.run_scan(&["--user"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("No skills found"));
}

#[test]
fn test_scan_json_output() {
    let ctx = ScanTestContext::new();
    
    ctx.create_skill(ctx.home(), ".claude/skills", "test-skill");
    
    let output = ctx.run_scan(&["--user", "--json"]);
    
    assert!(output.success());
    
    // Verify it's valid JSON
    let json: serde_json::Value = serde_json::from_str(&output.stdout)
        .expect("Output should be valid JSON");
    
    assert_eq!(json["skills_found"], 1);
    assert_eq!(json["valid"], 1);
}

#[test]
fn test_scan_verbose_output() {
    let ctx = ScanTestContext::new();
    
    ctx.create_skill(ctx.home(), ".claude/skills", "test-skill");
    
    let output = ctx.run_scan(&["--user", "--verbose"]);
    
    assert!(output.success());
    assert!(output.stdout.contains("test-skill"));
    assert!(output.stdout.contains("/home/.claude/skills/test-skill"));
}

```rust
use std::path::PathBuf;
use crate::paths::PathsConfig;
use crate::repo::RepoContext;
use crate::scanner::{scan_and_validate, ScanResult, OutputFormat};

pub struct ScanOptions {
    pub mode: ScanMode,
    pub dry_run: bool,
    pub verbose: bool,
    pub json: bool,  // Inherited from CLI
}

pub enum ScanMode {
    All,
    User,
    Repo,
    Tool(Vec<String>),
}

pub async fn execute(options: ScanOptions) -> Result<ScanResult, ScanError> {
    let paths_config = PathsConfig::load()?;
    let repo_context = RepoContext::discover().ok();
    
    let paths_to_scan = match options.mode {
        ScanMode::All => collect_all_paths(&paths_config, repo_context.as_ref()),
        ScanMode::User => collect_user_paths(&paths_config),
        ScanMode::Repo => {
            let repo = repo_context.ok_or(ScanError::NotInRepository)?;
            collect_repo_paths(&paths_config, &repo)
        }
        ScanMode::Tool(tools) => {
            validate_tools(&tools, &paths_config)?;
            collect_tool_paths(&paths_config, &tools, repo_context.as_ref())
        }
    };
    
    let format = if options.json {
        OutputFormat::Json
    } else if options.verbose {
        OutputFormat::Verbose
    } else {
        OutputFormat::Summary
    };
    
    scan_and_validate(&paths_to_scan, options.dry_run, format).await
}

fn collect_all_paths(
    config: &PathsConfig,
    repo: Option<&RepoContext>,
) -> Vec<PathBuf> {
    let mut paths = Vec::new();
    
    // Add repo paths if in a repo
    if let Some(repo) = repo {
        paths.extend(collect_repo_paths(config, repo));
    } else {
        log::info!("Not in a git repository, skipping repo scan");
    }
    
    // Add user paths
    paths.extend(collect_user_paths(config));
    
    paths
}

fn collect_user_paths(config: &PathsConfig) -> Vec<PathBuf> {
    // Expand all $HOME paths for all tools
    config.tools.values()
        .flat_map(|tool| config.expand_paths(tool, None))
        .collect()
}

fn collect_repo_paths(
    config: &PathsConfig,
    repo: &RepoContext,
) -> Vec<PathBuf> {
    // Expand $REPO_ROOT paths for all tools
    config.tools.values()
        .flat_map(|tool| config.expand_paths(tool, Some(&repo.root)))
        .filter(|path| repo.contains(path))
        .collect()
}

fn collect_tool_paths(
    config: &PathsConfig,
    tools: &[String],
    repo: Option<&RepoContext>,
) -> Vec<PathBuf> {
    let mut paths = Vec::new();
    
    for tool_name in tools {
        if let Some(tool) = config.get_tool(tool_name) {
            // Add repo paths if in a repo
            if let Some(repo) = repo {
                paths.extend(config.expand_paths(tool, Some(&repo.root)));
            }
            // Add home paths
            paths.extend(config.expand_paths(tool, None));
        }
    }
    
    paths
}

fn validate_tools(
    tools: &[String],
    config: &PathsConfig,
) -> Result<(), ScanError> {
    let invalid: Vec<_> = tools.iter()
        .filter(|t| !config.has_tool(t))
        .collect();
    
    if !invalid.is_empty() {
        return Err(ScanError::InvalidTools {
            invalid: invalid.into_iter().cloned().collect(),
            available: config.tool_names(),
        });
    }
    
    Ok(())
}

pub enum ScanError {
    NotInRepository,
    InvalidTools { invalid: Vec<String>, available: Vec<String> },
    ConfigError(crate::paths::ConfigError),
}
```

Commands::Scan(args) => {
    let mode = determine_mode(&args)?;
    let options = ScanOptions {
        mode,
        dry_run: args.dry_run,
        verbose: args.verbose,
        json: cli.json,
    };
    
    match scan::execute(options).await {
        Ok(result) => {
            let exit_code = determine_exit_code(&result);
            print_results(&result, cli.json);
            process::exit(exit_code);
        }
        Err(e) => {
            handle_scan_error(e, cli.json);
            process::exit(2);
        }
    }
}

use std::path::{Path, PathBuf};
use std::process::Command;
use tempfile::TempDir;

/// Test context for scan-related tests
pub struct ScanTestContext {
    pub temp_dir: TempDir,
    pub home_dir: PathBuf,
    pub repo_root: Option<PathBuf>,
}

impl ScanTestContext {
    /// Create a new test context
    pub fn new() -> Self {
        let temp_dir = TempDir::new().unwrap();
        let home_dir = temp_dir.path().join("home");
        std::fs::create_dir(&home_dir).unwrap();
        
        Self {
            temp_dir,
            home_dir,
            repo_root: None,
        }
    }
    
    /// Initialize a git repository in the temp directory
    pub fn init_git_repo(&mut self) -> &Path {
        let repo_root = self.temp_dir.path().join("repo");
        std::fs::create_dir(&repo_root).unwrap();
        
        Command::new("git")
            .args(&["init"])
            .current_dir(&repo_root)
            .output()
            .expect("Failed to init git repo");
        
        self.repo_root = Some(repo_root);
        self.repo_root.as_ref().unwrap()
    }
    
    /// Create a skill directory with SKILL.md
    pub fn create_skill(&self, base: &Path, tool: &str, name: &str) -> PathBuf {
        let skill_dir = base.join(tool).join("skills").join(name);
        std::fs::create_dir_all(&skill_dir).unwrap();
        
        let skill_content = format!(r#"---
name: {}
description: Test skill for {}
---

# {}

Test content.
"#, name, tool, name);
        
        std::fs::write(skill_dir.join("SKILL.md"), skill_content).unwrap();
        skill_dir
    }
    
    /// Create an invalid skill (missing required fields)
    pub fn create_invalid_skill(&self, base: &Path, tool: &str, name: &str) -> PathBuf {
        let skill_dir = base.join(tool).join("skills").join(name);
        std::fs::create_dir_all(&skill_dir).unwrap();
        
        // Missing description field
        let skill_content = format!(r#"---
name: {}
---

# {}

Test content.
"#, name, name);
        
        std::fs::write(skill_dir.join("SKILL.md"), skill_content).unwrap();
        skill_dir
    }
    
    /// Create a skill directory without SKILL.md (not a skill)
    pub fn create_non_skill_dir(&self, base: &Path, tool: &str, name: &str) -> PathBuf {
        let dir = base.join(tool).join("skills").join(name);
        std::fs::create_dir_all(&dir).unwrap();
        dir
    }
    
    /// Run the skills-validator command with given args
    pub fn run_scan(&self, args: &[&str]) -> CommandOutput {
        let mut cmd = Command::new("cargo");
        cmd.args(&["run", "--", "scan"]);
        cmd.args(args);
        cmd.env("HOME", &self.home_dir);
        
        if let Some(ref repo) = self.repo_root {
            cmd.current_dir(repo);
        }
        
        let output = cmd.output().expect("Failed to run command");
        
        CommandOutput {
            stdout: String::from_utf8_lossy(&output.stdout).to_string(),
            stderr: String::from_utf8_lossy(&output.stderr).to_string(),
            status: output.status,
        }
    }
    
    /// Get the home directory path
    pub fn home(&self) -> &Path {
        &self.home_dir
    }
    
    /// Get the repo root path (panics if not initialized)
    pub fn repo(&self) -> &Path {
        self.repo_root.as_ref().expect("Repository not initialized")
    }
}

/// Output from a command execution
pub struct CommandOutput {
    pub stdout: String,
    pub stderr: String,
    pub status: std::process::ExitStatus,
}

impl CommandOutput {
    pub fn success(&self) -> bool {
        self.status.success()
    }
    
    pub fn code(&self) -> Option<i32> {
        self.status.code()
    }
}

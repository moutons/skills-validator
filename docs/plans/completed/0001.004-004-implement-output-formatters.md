use serde::Serialize;
use std::io::Write;
use crate::scanner::{DuplicateSkill, ScanResult, ScannedSkill, OutputFormat};

/// Print scan results in the specified format
pub fn print_results(result: &ScanResult, format: OutputFormat, writer: &mut dyn Write) {
    match format {
        OutputFormat::Summary => print_summary(result, writer),
        OutputFormat::Verbose => print_verbose(result, writer),
        OutputFormat::Json => print_json(result, writer),
    }
}

/// Print summary format (default)
fn print_summary(result: &ScanResult, writer: &mut dyn Write) {
    let total = result.skills.len();
    let valid = result.valid_count();
    let invalid = result.invalid_count();
    let warnings = result.warning_count();
    
    if total == 0 {
        writeln!(writer, "⚠️  No skills found").unwrap();
        return;
    }
    
    writeln!(writer, "Found {} skill{} in {} location{}",
        total,
        if total == 1 { "" } else { "s" },
        result.skills.iter().map(|s| s.path.parent()).collect::<std::collections::HashSet<_>>().len(),
        if result.skills.len() == 1 { "" } else { "s" }
    ).unwrap();
    
    writeln!(writer, "Valid: {} | Invalid: {} | Warnings: {}",
        valid, invalid, warnings
    ).unwrap();
    
    if invalid > 0 {
        writeln!(writer).unwrap();
        writeln!(writer, "Invalid skills:").unwrap();
        for skill in &result.skills {
            if !skill.validation.errors.is_empty() {
                writeln!(writer, "  - {} ({})", skill.name, skill.path.display()).unwrap();
                for error in &skill.validation.errors {
                    writeln!(writer, "    Error: {}", error).unwrap();
                }
            }
        }
    }
    
    if !result.duplicates.is_empty() {
        writeln!(writer).unwrap();
        writeln!(writer, "⚠️  {} duplicate skill{} detected",
            result.duplicates.len(),
            if result.duplicates.len() == 1 { "" } else { "s" }
        ).unwrap();
        for dup in &result.duplicates {
            writeln!(writer, "  - {} exists in {} locations:", dup.name, dup.paths.len()).unwrap();
            for path in &dup.paths {
                writeln!(writer, "      {}", path.display()).unwrap();
            }
        }
        writeln!(writer, "   Your tool may not behave as expected. Check your tool's documentation.").unwrap();
    }
}

/// Print verbose format
fn print_verbose(result: &ScanResult, writer: &mut dyn Write) {
    if result.skills.is_empty() {
        writeln!(writer, "⚠️  No skills found").unwrap();
        return;
    }
    
    writeln!(writer, "Found {} skill(s):", result.skills.len()).unwrap();
    writeln!(writer).unwrap();
    
    for skill in &result.skills {
        let status = if skill.validation.errors.is_empty() {
            "✓"
        } else {
            "✗"
        };
        
        writeln!(writer, "{} {} ({})", status, skill.name, skill.path.display()).unwrap();
        
        if let Some(tool) = &skill.tool {
            writeln!(writer, "  Tool: {}", tool).unwrap();
        }
        
        if !skill.validation.errors.is_empty() {
            writeln!(writer, "  Errors:").unwrap();
            for error in &skill.validation.errors {
                writeln!(writer, "    - {}", error).unwrap();
            }
        }
        
        if !skill.validation.warnings.is_empty() {
            writeln!(writer, "  Warnings:").unwrap();
            for warning in &skill.validation.warnings {
                writeln!(writer, "    - {}", warning).unwrap();
            }
        }
        
        writeln!(writer).unwrap();
    }
    
    if !result.duplicates.is_empty() {
        writeln!(writer, "⚠️  Duplicates:").unwrap();
        for dup in &result.duplicates {
            writeln!(writer, "  {}: {} locations", dup.name, dup.paths.len()).unwrap();
        }
    }
}

/// JSON-serializable scan result
#[derive(Serialize)]
struct JsonScanResult<'a> {
    skills_found: usize,
    valid: usize,
    invalid: usize,
    warnings: usize,
    duplicates: usize,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    skills: Vec<JsonSkill<'a>>,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    duplicate_details: Vec<JsonDuplicate<'a>>,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    scan_errors: Vec<JsonScanError<'a>>,
}

#[derive(Serialize)]
struct JsonSkill<'a> {
    name: &'a str,
    path: String,
    tool: Option<&'a str>,
    valid: bool,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    errors: &'a [String],
    #[serde(skip_serializing_if = "Vec::is_empty")]
    warnings: &'a [String],
}

#[derive(Serialize)]
struct JsonDuplicate<'a> {
    name: &'a str,
    paths: Vec<String>,
}

#[derive(Serialize)]
struct JsonScanError<'a> {
    path: String,
    error: &'a str,
}

/// Print JSON format
fn print_json(result: &ScanResult, writer: &mut dyn Write) {
    let json_result = JsonScanResult {
        skills_found: result.skills.len(),
        valid: result.valid_count(),
        invalid: result.invalid_count(),
        warnings: result.warning_count(),
        duplicates: result.duplicates.len(),
        skills: result.skills.iter().map(|s| JsonSkill {
            name: &s.name,
            path: s.path.display().to_string(),
            tool: s.tool.as_deref(),
            valid: s.validation.errors.is_empty(),
            errors: &s.validation.errors,
            warnings: &s.validation.warnings,
        }).collect(),
        duplicate_details: result.duplicates.iter().map(|d| JsonDuplicate {
            name: &d.name,
            paths: d.paths.iter().map(|p| p.display().to_string()).collect(),
        }).collect(),
        scan_errors: result.scan_errors.iter().map(|e| JsonScanError {
            path: e.path.display().to_string(),
            error: &e.error,
        }).collect(),
    };
    
    writeln!(writer, "{}", serde_json::to_string_pretty(&json_result).unwrap()).unwrap();
}

#[cfg(test)]
mod tests {
    use super::*;
    
    fn create_test_result() -> ScanResult {
        ScanResult {
            skills: vec![
                ScannedSkill {
                    name: "valid-skill".to_string(),
                    path: std::path::PathBuf::from("/home/.claude/skills/valid-skill"),
                    validation: crate::validator::ValidationResult {
                        errors: vec![],
                        warnings: vec!["Consider adding examples".to_string()],
                    },
                    tool: Some("claude-code".to_string()),
                },
                ScannedSkill {
                    name: "invalid-skill".to_string(),
                    path: std::path::PathBuf::from("/home/.claude/skills/invalid-skill"),
                    validation: crate::validator::ValidationResult {
                        errors: vec!["Missing description".to_string()],
                        warnings: vec![],
                    },
                    tool: Some("claude-code".to_string()),
                },
            ],
            duplicates: vec![],
            scan_errors: vec![],
        }
    }
    
    #[test]
    fn test_print_summary() {
        let result = create_test_result();
        let mut output = Vec::new();
        print_summary(&result, &mut output);
        let text = String::from_utf8(output).unwrap();
        
        assert!(text.contains("Found 2 skills"));
        assert!(text.contains("Valid: 1"));
        assert!(text.contains("Invalid: 1"));
    }
    
    #[test]
    fn test_print_json() {
        let result = create_test_result();
        let mut output = Vec::new();
        print_json(&result, &mut output);
        let json: serde_json::Value = serde_json::from_slice(&output).unwrap();
        
        assert_eq!(json["skills_found"], 2);
        assert_eq!(json["valid"], 1);
        assert_eq!(json["invalid"], 1);
    }
}

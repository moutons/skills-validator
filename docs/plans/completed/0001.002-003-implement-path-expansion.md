## Implementation

Add to `src/paths.rs`:

```rust
use dirs::home_dir;

impl PathsConfig {
    /// Expand path templates for a specific tool
    /// 
    /// # Arguments
    /// * `tool_name` - The normalized tool name
    /// * `repo_root` - Optional repository root for $REPO_ROOT expansion
    /// 
    /// # Returns
    /// Vector of absolute paths that exist on the filesystem
    pub fn expand_paths(
        &self,
        tool_name: &str,
        repo_root: Option<&Path>,
    ) -> Vec<PathBuf> {
        let Some(tool) = self.get_tool(tool_name) else {
            return Vec::new();
        };
        
        self.expand_tool_paths(tool, repo_root)
    }
    
    /// Expand paths for a tool config
    pub fn expand_tool_paths(
        &self,
        tool: &ToolConfig,
        repo_root: Option<&Path>,
    ) -> Vec<PathBuf> {
        let home = home_dir();
        let mut result = Vec::new();
        
        for template in &tool.directories {
            // Skip paths requiring repo if no repo provided
            if template.requires_repo && repo_root.is_none() {
                continue;
            }
            
            let expanded = expand_path_template(&template.template, home.as_deref(), repo_root);
            
            if let Some(path) = expanded {
                // Only include paths that actually exist
                if path.exists() {
                    result.push(path);
                }
            }
        }
        
        result
    }
    
    /// Expand all paths for all tools
    pub fn expand_all_paths(&self, repo_root: Option<&Path>) -> Vec<PathBuf> {
        self.tools.values()
            .flat_map(|tool| self.expand_tool_paths(tool, repo_root))
            .collect()
    }
}

/// Expand a single path template
fn expand_path_template(
    template: &str,
    home: Option<&Path>,
    repo_root: Option<&Path>,
) -> Option<PathBuf> {
    let mut result = template.to_string();
    
    // Expand $HOME
    if result.contains("$HOME") {
        match home {
            Some(home_path) => {
                result = result.replace("$HOME", &home_path.to_string_lossy());
            }
            None => {
                log::warn!("$HOME not available, skipping path: {}", template);
                return None;
            }
        }
    }
    
    // Expand $REPO_ROOT
    if result.contains("$REPO_ROOT") {
        match repo_root {
            Some(repo) => {
                result = result.replace("$REPO_ROOT", &repo.to_string_lossy());
            }
            None => {
                log::debug!("$REPO_ROOT not available, skipping path: {}", template);
                return None;
            }
        }
    }
    
    Some(PathBuf::from(result))
}

/// Find all skill directories within a path
/// 
/// Returns paths that contain a SKILL.md or skill.md file
pub fn find_skill_directories(root: &Path) -> Vec<PathBuf> {
    let mut result = Vec::new();
    
    if !root.is_dir() {
        return result;
    }
    
    if let Ok(entries) = std::fs::read_dir(root) {
        for entry in entries.flatten() {
            let path = entry.path();
            if path.is_dir() {
                // Check if this directory contains a skill file
                let skill_md = path.join("SKILL.md");
                let skill_md_lower = path.join("skill.md");
                
                if skill_md.exists() || skill_md_lower.exists() {
                    result.push(path);
                }
            }
        }
    }
    
    result
}
```

## Testing

```rust
#[cfg(test)]
mod tests {
    use super::*;
    use std::io::Write;
    use tempfile::TempDir;
    
    #[test]
    fn test_expand_home() {
        let template = "$HOME/.claude/skills";
        let home = Some(Path::new("/home/user"));
        let result = expand_path_template(template, home, None);
        assert_eq!(result, Some(PathBuf::from("/home/user/.claude/skills")));
    }
    
    #[test]
    fn test_expand_repo() {
        let template = "$REPO_ROOT/.claude/skills";
        let repo = Some(Path::new("/repo"));
        let result = expand_path_template(template, None, repo);
        assert_eq!(result, Some(PathBuf::from("/repo/.claude/skills")));
    }
    
    #[test]
    fn test_find_skill_directories() {
        let temp = TempDir::new().unwrap();
        let root = temp.path();
        
        // Create a skill directory
        let skill_dir = root.join("my-skill");
        std::fs::create_dir(&skill_dir).unwrap();
        let mut file = std::fs::File::create(skill_dir.join("SKILL.md")).unwrap();
        writeln!(file, "# My Skill").unwrap();
        
        // Create a non-skill directory
        std::fs::create_dir(root.join("not-a-skill")).unwrap();
        
        let found = find_skill_directories(root);
        assert_eq!(found.len(), 1);
        assert!(found[0].ends_with("my-skill"));
    }
}
```

use std::collections::HashMap;
use std::path::{Path, PathBuf};
use serde::Deserialize;

/// Raw configuration structure matching paths.jsonc
#[derive(Debug, Deserialize)]
struct RawToolConfig {
    name: String,
    #[serde(default)]
    documentation: Option<String>,
    directories: Vec<String>,
}

#[derive(Debug, Deserialize)]
struct RawPathsConfig {
    #[serde(flatten)]
    tools: HashMap<String, RawToolConfig>,
    #[serde(rename = "_unsupported")]
    unsupported: Option<HashMap<String, RawToolConfig>>,
}

/// Parsed tool configuration
#[derive(Debug, Clone)]
pub struct ToolConfig {
    pub name: String,
    pub documentation: Option<String>,
    pub directories: Vec<PathTemplate>,
}

/// A path template that may contain variables
#[derive(Debug, Clone)]
pub struct PathTemplate {
    pub template: String,
    pub requires_repo: bool,
}

/// Main configuration container
#[derive(Debug)]
pub struct PathsConfig {
    tools: HashMap<String, ToolConfig>,
}

#[derive(Debug, thiserror::Error)]
pub enum ConfigError {
    #[error("Failed to parse paths.jsonc: {0}")]
    ParseError(String),
}

impl PathsConfig {
    /// Load configuration from embedded paths.jsonc
    pub fn load() -> Result<Self, ConfigError> {
        const RAW_CONFIG: &str = include_str!("../paths.jsonc");
        
        // Strip JSONC comments before parsing
        let json = strip_jsonc_comments(RAW_CONFIG);
        
        let raw: RawPathsConfig = serde_json::from_str(&json)
            .map_err(|e| ConfigError::ParseError(e.to_string()))?;
        
        let tools = raw.tools.into_iter()
            .map(|(key, config)| {
                let directories = config.directories.into_iter()
                    .map(|template| PathTemplate {
                        requires_repo: template.contains("$REPO_ROOT"),
                        template,
                    })
                    .collect();
                
                let tool = ToolConfig {
                    name: config.name,
                    documentation: config.documentation,
                    directories,
                };
                
                (key, tool)
            })
            .collect();
        
        Ok(Self { tools })
    }
    
    /// Get tool configuration by normalized name
    pub fn get_tool(&self, name: &str) -> Option<&ToolConfig> {
        // Try exact match first
        if let Some(tool) = self.tools.get(name) {
            return Some(tool);
        }
        
        // Try normalized (kebab-case) match
        let normalized = normalize_tool_name(name);
        self.tools.get(&normalized)
    }
    
    /// Check if a tool exists
    pub fn has_tool(&self, name: &str) -> bool {
        self.get_tool(name).is_some()
    }
    
    /// List all available tool names (normalized keys)
    pub fn tool_names(&self) -> Vec<&str> {
        self.tools.keys().map(|k| k.as_str()).collect()
    }
    
    /// Get all tool configs
    pub fn tools(&self) -> &HashMap<String, ToolConfig> {
        &self.tools
    }
}

/// Normalize a tool name to kebab-case
fn normalize_tool_name(name: &str) -> String {
    name.to_lowercase()
        .replace(' ', "-")
        .replace('_', "-")
}

/// Strip JSONC comments, returning valid JSON
fn strip_jsonc_comments(input: &str) -> String {
    // Simple implementation: remove // comments and /* */ blocks
    // For production, consider using a proper JSONC parser crate
    let mut result = String::new();
    let mut chars = input.chars().peekable();
    
    while let Some(ch) = chars.next() {
        match ch {
            '/' => {
                match chars.peek() {
                    Some('/') => {
                        // Line comment, skip to end of line
                        while chars.next().map_or(false, |c| c != '\n') {}
                    }
                    Some('*') => {
                        // Block comment, skip to */
                        chars.next(); // consume *
                        loop {
                            match chars.next() {
                                Some('*') => {
                                    if chars.peek() == Some(&'/') {
                                        chars.next();
                                        break;
                                    }
                                }
                                None => break,
                                _ => {}
                            }
                        }
                    }
                    _ => result.push(ch),
                }
            }
            _ => result.push(ch),
        }
    }
    
    result
}

## Testing

Create unit tests in the module:

```rust
#[cfg(test)]
mod tests {
    use super::*;
    
    #[test]
    fn test_load_config() {
        let config = PathsConfig::load().unwrap();
        assert!(!config.tools.is_empty());
        assert!(config.has_tool("claude-code"));
    }
    
    #[test]
    fn test_normalize_tool_name() {
        assert_eq!(normalize_tool_name("Claude Code"), "claude-code");
        assert_eq!(normalize_tool_name("claude_code"), "claude-code");
        assert_eq!(normalize_tool_name("CLAUDE-CODE"), "claude-code");
    }
    
    #[test]
    fn test_strip_comments() {
        let jsonc = r#"{
            // This is a comment
            "key": "value", /* block comment */
            "num": 42
        }"#;
        let json = strip_jsonc_comments(jsonc);
        assert!(!json.contains("//"));
        assert!(!json.contains("/*"));
        assert!(json.contains("\"key\""));
    }
}
```

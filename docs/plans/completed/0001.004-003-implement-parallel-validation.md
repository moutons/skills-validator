```rust
use rayon::prelude::*;
use std::path::Path;
use crate::parser::read_properties;
use crate::scanner::{ScanResult, ScannedSkill, DuplicateSkill, ScanError};
use crate::validator::validate;

/// Validate all discovered skills in parallel
/// 
/// If `dry_run` is true, skips validation and marks all skills as valid
pub fn validate_skills_parallel(
    mut skills: Vec<ScannedSkill>,
    dry_run: bool,
) -> Vec<ScannedSkill> {
    if dry_run {
        // In dry-run mode, just mark all as valid without actually validating
        for skill in &mut skills {
            skill.validation.errors.clear();
            skill.validation.warnings.clear();
        }
        return skills;
    }
    
    // Use rayon's parallel iterator for concurrent validation
    skills.par_iter_mut().for_each(|skill| {
        match validate_skill(skill) {
            Ok(result) => skill.validation = result,
            Err(e) => {
                skill.validation.errors.push(format!("Validation failed: {}", e));
            }
        }
    });
    
    skills
}

/// Validate a single skill
fn validate_skill(skill: &ScannedSkill) -> Result<crate::validator::ValidationResult, String> {
    // Use existing validate function from validator module
    Ok(validate(&skill.path))
}

/// Main entry point: discover and validate skills
pub fn scan_and_validate(
    paths: &[std::path::PathBuf],
    config: &crate::paths::PathsConfig,
    dry_run: bool,
) -> ScanResult {
    // First, discover all skills
    let (mut skills, scan_errors) = crate::scanner::discovery::discover_skills(paths, config);
    
    // Validate in parallel
    skills = validate_skills_parallel(skills, dry_run);
    
    // Detect duplicates
    let duplicates = crate::scanner::discovery::detect_duplicates(&skills);
    
    ScanResult {
        skills,
        duplicates,
        scan_errors,
    }
}

/// Determine exit code based on scan results
/// 
/// Returns:
/// - 0: All skills valid (warnings OK)
/// - 1: Some skills invalid
/// - 2: Only scan errors (no skills found or system errors)
pub fn determine_exit_code(result: &ScanResult) -> i32 {
    // If there are scan errors but no skills, that's a system error
    if result.skills.is_empty() {
        if result.scan_errors.is_empty() {
            // No skills found, but no errors either - this is a warning scenario
            // Return 0 but caller should show warning
            return 0;
        }
        return 2;
    }
    
    // Check for validation errors
    let invalid_count = result.skills.iter()
        .filter(|s| !s.validation.errors.is_empty())
        .count();
    
    if invalid_count > 0 {
        return 1;
    }
    
    0
}
```

## Testing

```rust
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;
    
    #[test]
    fn test_validate_skills_parallel() {
        let skills = vec![
            ScannedSkill {
                name: "valid-skill".to_string(),
                path: PathBuf::from("/some/path"),
                validation: Default::default(),
                tool: None,
            },
        ];
        
        // Test dry run
        let result = validate_skills_parallel(skills.clone(), true);
        assert!(result[0].validation.errors.is_empty());
    }
    
    #[test]
    fn test_determine_exit_code() {
        let mut result = ScanResult::new();
        
        // Empty result
        assert_eq!(determine_exit_code(&result), 0);
        
        // With scan error
        result.scan_errors.push(ScanError {
            path: PathBuf::from("/bad"),
            error: "Permission denied".to_string(),
        });
        assert_eq!(determine_exit_code(&result), 2);
        
        // With valid skill
        result.skills.push(ScannedSkill {
            name: "valid".to_string(),
            path: PathBuf::from("/good"),
            validation: crate::validator::ValidationResult {
                errors: vec![],
                warnings: vec![],
            },
            tool: None,
        });
        assert_eq!(determine_exit_code(&result), 0);
        
        // With invalid skill
        result.skills.push(ScannedSkill {
            name: "invalid".to_string(),
            path: PathBuf::from("/bad"),
            validation: crate::validator::ValidationResult {
                errors: vec!["Missing field".to_string()],
                warnings: vec![],
            },
            tool: None,
        });
        assert_eq!(determine_exit_code(&result), 1);
    }
}
```

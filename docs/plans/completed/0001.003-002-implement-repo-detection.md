```rust
use git2::Repository;
use std::path::{Path, PathBuf};

/// Context for a git repository
#[derive(Debug, Clone)]
pub struct RepoContext {
    /// The repository root directory
    pub root: PathBuf,
    /// The current working directory when discovered
    pub current_dir: PathBuf,
}

#[derive(Debug, thiserror::Error)]
pub enum RepoError {
    #[error("Not in a git repository")]
    NotARepository,
    
    #[error("Git error: {0}")]
    GitError(#[from] git2::Error),
}

impl RepoContext {
    /// Discover git repository from current working directory
    pub fn discover() -> Result<Self, RepoError> {
        let current_dir = std::env::current_dir()
            .map_err(|e| RepoError::GitError(git2::Error::from_str(&e.to_string())))?;
        
        Self::from_path(&current_dir)
    }
    
    /// Discover git repository from specific path
    pub fn from_path(path: &Path) -> Result<Self, RepoError> {
        let repo = Repository::discover(path)?;
        let root = repo.workdir()
            .ok_or_else(|| RepoError::GitError(
                git2::Error::from_str("Repository has no working directory")
            ))?;
        
        Ok(Self {
            root: root.to_path_buf(),
            current_dir: path.to_path_buf(),
        })
    }
    
    /// Check if a path is within this repository
    pub fn contains(&self, path: &Path) -> bool {
        path.starts_with(&self.root)
    }
    
    /// Get the relative path from repo root to given path
    pub fn relative_to_root(&self, path: &Path) -> Option<PathBuf> {
        path.strip_prefix(&self.root).ok().map(|p| p.to_path_buf())
    }
}

/// Get all directories from current dir up to (and including) repo root
/// 
/// For example, if CWD is /repo/src/components and root is /repo,
/// returns: [/repo/src/components, /repo/src, /repo]
pub fn directories_to_root(repo: &RepoContext) -> Vec<PathBuf> {
    let mut result = Vec::new();
    let mut current = repo.current_dir.clone();
    
    loop {
        result.push(current.clone());
        
        if current == repo.root {
            break;
        }
        
        match current.parent() {
            Some(parent) => current = parent.to_path_buf(),
            None => break,
        }
    }
    
    result
}

/// Check if current directory is in a git repository
pub fn is_in_repository() -> bool {
    RepoContext::discover().is_ok()
}
```

## Testing

```rust
#[cfg(test)]
mod tests {
    use super::*;
    use std::process::Command;
    use tempfile::TempDir;
    
    fn create_git_repo(path: &Path) {
        Command::new("git")
            .args(&["init"])
            .current_dir(path)
            .output()
            .expect("Failed to init git repo");
    }
    
    #[test]
    fn test_discover_repo() {
        let temp = TempDir::new().unwrap();
        create_git_repo(temp.path());
        
        let context = RepoContext::from_path(temp.path()).unwrap();
        assert_eq!(context.root, temp.path());
    }
    
    #[test]
    fn test_not_a_repo() {
        let temp = TempDir::new().unwrap();
        
        let result = RepoContext::from_path(temp.path());
        assert!(matches!(result, Err(RepoError::NotARepository)));
    }
    
    #[test]
    fn test_directories_to_root() {
        let temp = TempDir::new().unwrap();
        create_git_repo(temp.path());
        
        // Create nested directory
        let nested = temp.path().join("src/components");
        std::fs::create_dir_all(&nested).unwrap();
        
        let context = RepoContext {
            root: temp.path().to_path_buf(),
            current_dir: nested.clone(),
        };
        
        let dirs = directories_to_root(&context);
        assert_eq!(dirs.len(), 3);
        assert!(dirs[0].ends_with("components"));
        assert!(dirs[1].ends_with("src"));
        assert_eq!(dirs[2], temp.path());
    }
}
```

use std::collections::HashMap;
use std::path::{Path, PathBuf};
use crate::paths::{find_skill_directories, PathsConfig, ToolConfig};
use crate::scanner::{ScanError, ScannedSkill};

/// Discover all skills in the given paths
/// 
/// For each path, finds all skill directories and returns ScannedSkill structs
/// (without validation results yet)
pub fn discover_skills(
    paths: &[PathBuf],
    config: &PathsConfig,
) -> (Vec<ScannedSkill>, Vec<ScanError>) {
    let mut skills = Vec::new();
    let mut errors = Vec::new();
    let mut seen_paths = HashMap::new();
    
    for path in paths {
        if !path.exists() {
            // Skip non-existent paths silently
            continue;
        }
        
        if !path.is_dir() {
            errors.push(ScanError {
                path: path.clone(),
                error: "Not a directory".to_string(),
            });
            continue;
        }
        
        // Find which tool this path belongs to (if any)
        let tool = find_tool_for_path(path, config);
        
        // Find skill directories within this path
        match discover_skills_in_directory(path, tool.as_deref()) {
            Ok(found) => {
                for skill in found {
                    // Track canonical path to detect duplicates
                    match std::fs::canonicalize(&skill.path) {
                        Ok(canonical) => {
                            seen_paths.entry(canonical.clone())
                                .or_insert_with(Vec::new)
                                .push(skill.path.clone());
                        }
                        Err(e) => {
                            log::warn!("Could not canonicalize {}: {}", skill.path.display(), e);
                        }
                    }
                    
                    skills.push(skill);
                }
            }
            Err(e) => {
                errors.push(ScanError {
                    path: path.clone(),
                    error: e.to_string(),
                });
            }
        }
    }
    
    (skills, errors)
}

/// Discover skills within a single directory
fn discover_skills_in_directory(
    dir: &Path,
    tool: Option<&str>,
) -> Result<Vec<ScannedSkill>, std::io::Error> {
    let mut skills = Vec::new();
    
    for skill_dir in find_skill_directories(dir) {
        // Extract skill name from directory name
        let name = skill_dir.file_name()
            .and_then(|n| n.to_str())
            .unwrap_or("unknown")
            .to_string();
        
        skills.push(ScannedSkill {
            name,
            path: skill_dir,
            validation: crate::validator::ValidationResult::default(),
            tool: tool.map(|t| t.to_string()),
        });
    }
    
    Ok(skills)
}

/// Find which tool configuration a path belongs to
fn find_tool_for_path(path: &Path, config: &PathsConfig) -> Option<String> {
    let path_str = path.to_string_lossy();
    
    for (tool_key, tool_config) in config.tools() {
        for dir_template in &tool_config.directories {
            // Simple check: does the path contain the tool name?
            // More sophisticated: expand template and compare
            if path_str.contains(tool_key) || 
               path_str.contains(&tool_config.name.to_lowercase()) {
                return Some(tool_key.clone());
            }
        }
    }
    
    None
}

/// Detect duplicate skills by name
pub fn detect_duplicates(skills: &[ScannedSkill]) -> Vec<crate::scanner::DuplicateSkill> {
    let mut by_name: HashMap<String, Vec<PathBuf>> = HashMap::new();
    
    for skill in skills {
        by_name.entry(skill.name.clone())
            .or_default()
            .push(skill.path.clone());
    }
    
    by_name.into_iter()
        .filter(|(_, paths)| paths.len() > 1)
        .map(|(name, paths)| crate::scanner::DuplicateSkill { name, paths })
        .collect()
}

#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::TempDir;
    use std::io::Write;
    
    fn create_skill_dir(parent: &Path, name: &str) -> PathBuf {
        let dir = parent.join(name);
        std::fs::create_dir(&dir).unwrap();
        let mut file = std::fs::File::create(dir.join("SKILL.md")).unwrap();
        writeln!(file, "# {}").unwrap();
        dir
    }
    
    #[test]
    fn test_discover_skills() {
        let temp = TempDir::new().unwrap();
        let skills_dir = temp.path().join(".claude/skills");
        std::fs::create_dir_all(&skills_dir).unwrap();
        
        create_skill_dir(&skills_dir, "skill-a");
        create_skill_dir(&skills_dir, "skill-b");
        
        let config = PathsConfig::load().unwrap();
        let (skills, errors) = discover_skills(&[skills_dir], &config);
        
        assert_eq!(skills.len(), 2);
        assert!(errors.is_empty());
        assert!(skills.iter().any(|s| s.name == "skill-a"));
        assert!(skills.iter().any(|s| s.name == "skill-b"));
    }
    
    #[test]
    fn test_detect_duplicates() {
        let skills = vec![
            ScannedSkill {
                name: "my-skill".to_string(),
                path: PathBuf::from("/home/user/.claude/skills/my-skill"),
                validation: Default::default(),
                tool: None,
            },
            ScannedSkill {
                name: "my-skill".to_string(),
                path: PathBuf::from("/repo/.claude/skills/my-skill"),
                validation: Default::default(),
                tool: None,
            },
            ScannedSkill {
                name: "other-skill".to_string(),
                path: PathBuf::from("/home/user/.claude/skills/other-skill"),
                validation: Default::default(),
                tool: None,
            },
        ];
        
        let dups = detect_duplicates(&skills);
        assert_eq!(dups.len(), 1);
        assert_eq!(dups[0].name, "my-skill");
        assert_eq!(dups[0].paths.len(), 2);
    }
}
